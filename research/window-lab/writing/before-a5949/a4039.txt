para : column
  role block
  let talked = "yes" when $d.design != "" and $d.talk != "" and count(lines:$d.talk about=$item) > 0
  let talked = "no" otherwise
  let spots = join($quote of lines:$d.talk about=$item) when talked = "yes"
  let spots = "" otherwise
  let addressee = "@$d.by: " when $d.by != ""
  let addressee = "" otherwise
  let talkable = "yes" when $d.design != "" and $d.talk != "" and $r.view != "margins"
  let talkable = "no" otherwise
  let held = $r.quote when talkable = "yes" and $r.quoting = $item
  let held = "" otherwise
  on select when talkable = "yes" do prop $r quoting=$item quote="$value"
  on menu "Comment" when talkable = "no" do comment $item line="$(addressee)$selected$value"
  on menu "Unclear to me" when talkable = "no" do comment $item line="$(addressee)$selected unclear to me"
  on menu "I agree" when talkable = "no" do comment $item line="$(addressee)$selected I agree"
  on menu "I disagree" when talkable = "no" do comment $item line="$(addressee)$selected I disagree"
  on menu "Comment" when talkable = "yes" do append $d.talk line="$value" about=$item quote="$(held)"; do prop $r quote=""
  on menu "Unclear to me" when talkable = "yes" do append $d.talk line="unclear to me" about=$item quote="$(held)" mark=ask; do prop $r quote=""
  on menu "I agree" when talkable = "yes" do append $d.talk line="I agree" about=$item quote="$(held)"; do prop $r quote=""
  on menu "I disagree" when talkable = "yes" do append $d.talk line="I disagree" about=$item quote="$(held)"; do prop $r quote=""
  let shape = "h2" when $block = "h2"
  let shape = "bullet" when $block = "bullet"
  let shape = "number" when $block = "number"
  let shape = "check" when $block = "check"
  let shape = "p" otherwise
  on menu "Plain text" when shape != "p" do prop $item remove=block
  on menu "Heading" when shape != "h2" do prop $item block=h2
  on menu "Bullet list" when shape != "bullet" do prop $item block=bullet
  on menu "Numbered list" when shape != "number" do prop $item block=number
  on menu "Check list" when shape != "check" do prop $item block=check
  on menu "Move up" do up $item
  on menu "Move down" do down $item
  on menu "Delete line" do remove $item
  let ticked = "yes" when $checked = "yes"
  let ticked = "no" otherwise
  let drawn = "yes" when $block = "figure"
  let drawn = "no" otherwise
  let next = "bullet" when $block = "bullet"
  let next = "number" when $block = "number"
  let next = "check" when $block = "check"
  let next = "p" otherwise
  let ind = 3 when $indent = 3
  let ind = 2 when $indent = 2
  let ind = 1 when $indent = 1
  let ind = 0 otherwise
  let more = ind + 1 when ind < 3
  let more = 3 otherwise
  let less = ind - 1 when ind > 0
  let less = 0 otherwise
  line : row
    gap
    role in1 when ind = 1
    role in2 when ind = 2
    role in3 when ind = 3
    dot : text "•"
      role bullet
      only when $block = "bullet"
    num : text "$(place)."
      role bullet
      only when $block = "number"
    box : text "☐"
      role box
      only when $block = "check" and ticked = "no"
      on click do prop $item checked=yes
    tick : text "☑"
      role box.on
      only when $block = "check" and ticked = "yes"
      on click do prop $item remove=checked
    pic : text "$text"
      share 1
      role figure
      picture
      only when $block = "figure"
    words : text "$text"
      share 1
      only when drawn = "no" and $item != $r.reviewing
      role body
      role h1 when $block = "h1"
      role h2 when $block = "h2"
      role h3 when $block = "h3"
      role quote when $block = "quote"
      role ticked when $block = "check" and ticked = "yes"
      role item when $block = "bullet" or $block = "number" or $block = "check"
      marks
      light "$(spots)" "$(held)"
      on edit do rewrite $item text="$value"
      on key Enter do split $item $(next) "$head" "$tail"
      let previouswords = join($text of $before) or "" when $before != ""
      let previouswords = "" otherwise
      let previouskind = join($block of $before) or "p" when $before != ""
      let previouskind = "" otherwise
      on key Backspace when ind > 0 do prop $item indent=$(less)
      on key Backspace when ind = 0 and shape != "p" do prop $item remove=block
      on key Backspace when ind = 0 and shape = "p" and $before != "" and previouskind != "figure" and previouskind != "divider" do joinwritten $before $item "$(previouswords)" "$head$tail"
      let followingwords = join($text of $after) or "" when $after != ""
      let followingwords = "" otherwise
      let followingkind = join($block of $after) or "p" when $after != ""
      let followingkind = "" otherwise
      on key Delete when $after != "" and followingkind != "figure" and followingkind != "divider" do joinwritten $item $after "$head$tail" "$(followingwords)"
      on key Tab do prop $item indent=$(more)
      on key Shift+Tab do prop $item indent=$(less)
    nativewords : text "$text"
      commit ctrl-enter
      share 1
      only when drawn = "no" and $item = $r.reviewing
      role body
      role h1 when $block = "h1"
      role h2 when $block = "h2"
      role h3 when $block = "h3"
      role quote when $block = "quote"
      role ticked when $block = "check" and ticked = "yes"
      role item when $block = "bullet" or $block = "number" or $block = "check"
      marks
      light "$(spots)" "$(held)"
      on edit do rewrite $item text="$value"
    pip : text "●"
      role pip
      only when talked = "yes"
