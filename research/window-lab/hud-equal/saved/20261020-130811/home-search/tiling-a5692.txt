# home-tiling: the home base's panels, laid out by each person (a5496; made to work by a5994). Alex, 16 Oct: 'I need
# things to be draggable around this screen. It should snap to various sizes as I move it around, pushing other things
# out of the way ... Double-clicking a window should make it fill the screen or the dominant space ... while leaving
# the chat here.' Where each panel is, is the viewer's own world data: a surface (role=home-layout, actor=$me) holding
# column surfaces (span=1|2|3, its share of the width), each holding panel notes (role=home-panel): lens= what it shows,
# lesser= its one-line view, shelf=yes put away. On the layout: full= the panel that fills the space, pin=left|right
# the chat's side (right when unsaid).
# Drag a panel by its title. Over another panel, the landing box shows where it goes: near its left or right edge, a
# new column there; near its top or bottom, above or below it; its middle, the two trade places. Over a column's empty
# foot, it goes last there and takes that room (near the foot's left or right edge, a new column there). A column left empty goes, and the others take its width. Each drop is
# one use of the verb drop-panel (Home base). ⇔ widens a column: a third, a half, two thirds of the room. A
# double-click on a title fills the room with that panel (the others become a rail of titles; the chat stays); again,
# or ⤡, gives it back. – shelves it (home-panels, in the dock). Nothing scrolls, nothing moves by itself.
stage : column
  share 1
  role stage
  of prop:role=home-layout,prop:actor=$me as lay
  each desk
  lost : text "No layout of yours here: a surface in Home base with role=home-layout actor=<you>, holding column surfaces of panel notes (research/window-lab/home-layout/layout-alex.desk makes Alex's)."
    role hint
    only when count(prop:role=home-layout,prop:actor=$me) = 0
desk : column
  share 1
  role stage
  top : row
    role top
    widgets : row
      lens home-widgets
    brand : text "AGORA"
      role brand
    gap : column
      share 1
    how : text "drag a title anywhere · double-click a title to fill the room · ⇔ widens · – shelves it"
      role hint
    side : text "chat ⇆"
      role tool
      let at = ($lay.pin or "right")
      on click when ($lay.pin or "right") = "right" do prop $lay pin=left
      on click when ($lay.pin or "right") = "left" do prop $lay pin=right
    keys : text "Ctrl+Alt+J or Esc closes"
      role meta
  body : row
    share 1
    role body
    scroll none
    chatleft : column
      let wide = 7 when ($lay.full or "") != ""
      let wide = 32 when ($lay.full or "") = "" and count(prop:role=home-view,prop:actor=$me,prop:viewing,-prop:viewing=a5080) + count(prop:role=home-view,prop:actor=$me,prop:find) > 0
      let wide = 7 otherwise
      share $wide
      role pinned
      scroll none
      only when ($lay.pin or "right") = "left"
      lens home-viewer
    tiling : row
      share 13
      role tiling
      scroll none
      only when ($lay.full or "") = ""
      of kind:surface,in:$lay.item as col
      each column
    filled : row
      share 13
      role tiling
      scroll none
      only when ($lay.full or "") != ""
      rail : column
        role rail
        scroll none
        of kind:note,in:$lay.item,prop:role=home-panel,prop:shelf=no as r
        each railitem
      room : column
        share 1
        role tilecol
        scroll none
        of kind:note,in:$lay.item,prop:role=home-panel,prop:shelf=no as p
        each fullpanel
    chatright : column
      let wide = 7 when ($lay.full or "") != ""
      let wide = 32 when ($lay.full or "") = "" and count(prop:role=home-view,prop:actor=$me,prop:viewing,-prop:viewing=a5080) + count(prop:role=home-view,prop:actor=$me,prop:find) > 0
      let wide = 7 otherwise
      share $wide
      role pinned
      scroll none
      only when ($lay.pin or "right") != "left"
      lens home-viewer
column : column
  let span = ($col.span or 1)
  share $span
  role tilecol
  scroll none
  only when count(kind:note,in:$col.item,prop:role=home-panel,prop:shelf=no) > 0
  of kind:note,in:$col.item,prop:role=home-panel,prop:shelf=no as p
  each panel
  foot : column
    share 1
    role tilefoot
    on drop grip at left do drop-panel left $carried $col $col $lay
    on drop grip at right do drop-panel right $carried $col $col $lay
    on drop grip do drop-panel end $carried $col $col $lay
panel : column
  share 1
  role pane
  scroll none
  on drop grip at left do drop-panel left $carried $p $col $lay
  on drop grip at right do drop-panel right $carried $p $col $lay
  on drop grip at top do drop-panel top $carried $p $col $lay
  on drop grip at bottom do drop-panel bottom $carried $p $col $lay
  on drop grip do drop-panel swap $carried $p $col $lay
  grip : row
    role panehead
    carry
    let was = ($lay.full or "")
    on double when ($lay.full or "") = "" do prop $lay full=$p
    title : text "$title"
      role panetitle
    sum : column
      role panesum
      share 1
      only when ($p.lesser or "") != ""
      lens $p.lesser
    gap : column
      share 1
      only when ($p.lesser or "") = ""
    wider : text "⇔"
      role tool
      let next = 2 when ($col.span or 1) = 1
      let next = 3 when ($col.span or 1) = 2
      let next = 1 otherwise
      on click do prop $col span=$next
    away : text "–"
      role tool
      let out = ($shelf or "")
      on click do prop $p shelf=yes
  body : column
    share 1
    scroll scroll
    role panebody
    lens $p.lens
railitem : column
  role railitem
  role railitem.on when $item = ($lay.full or "")
  let was = ($lay.full or "")
  on click when $item != ($lay.full or "") do prop $lay full=$r
  name : text "$title"
    role chipname
    clamp 1
  sum : column
    role chipsum
    only when ($r.lesser or "") != ""
    lens $r.lesser
fullpanel : column
  share 1
  role pane
  scroll none
  only when $item = ($lay.full or "")
  grip : row
    role panehead
    let was = ($lay.full or "")
    on double when ($lay.full or "") != "" do prop $lay full=""
    title : text "$title"
      role panetitle
    gap : column
      share 1
    back : text "⤡"
      role tool
      let was = ($lay.full or "")
      on click when ($lay.full or "") != "" do prop $lay full=""
    away : text "–"
      role tool
      let out = ($shelf or "")
      on click do prop $p shelf=yes; do prop $lay full=""
  body : column
    role panebody
    share 1
    scroll scroll
    lens $p.lens
