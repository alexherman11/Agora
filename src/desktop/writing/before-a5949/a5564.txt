# home-dock: the dock along the foot of the home base (a5500), where Alex's taskbar was. Its icons are not lines here:
# they are the notes in his dock (the surface with role=dock actor=$me in the Home base room, f5204), in its order.
# A note's title is the label and glyph= its icon. opens=<id> shows that thing in the middle (home-viewer); opens=chat
# gives JEFF's chat back; window=<lens> opens a lens as its own window (the home base's host verb `window`).
# Add one: press + while something is open in the middle, or
#   do create <the dock> kind=note title="Label" glyph=◆ opens=<id>
# Right-click an icon to take it off the dock or move it along. The icon of what is open now is lit.
foot : row
  share 1
  role appfoot
  left : column
    share 1
  dock : row
    role appdock
    of kind:surface,prop:role=dock,prop:actor=$me as k
    each shelf
    none : text "No dock yet: make a row surface in the Home base room (f5204) and give it role=dock actor=<you>"
      role hint
      only when count(kind:surface,prop:role=dock,prop:actor=$me) = 0
  right : column
    share 1
    panels : row
      lens home-panels
    # a5496's panel toggles go here once their lens exists (a lens named here must exist, or this one stops compiling):
    #   panels : row
    #     lens home-panels
shelf : row
  share 1
  role appshelf
  of prop:role=home-view,prop:actor=$me as v
  each icons
icons : row
  share 1
  role appshelf
  apps : row
    role appicons
    of in:$k.item,kind:note as d
    each app
  add : row
    role appicons
    only when ($v.viewing or "") != "" and count(in:$k.item,prop:opens=$v.viewing) = 0
    of $v.viewing as r
    each adder
app : column
  role appslot
  on menu "Take off the dock" do remove $d.item
  on menu "Move left" when ($d.before or "") != "" do move $d.item before=$d.before
  on menu "Move right" when ($d.after or "") != "" do move $d.item after=$d.after
  view : column
    only when ($d.opens or "") != "" and $d.opens != "chat"
    role appicon
    role appicon.on when $d.opens = $v.viewing
    on click do prop $v viewing=$d.opens back="$v.viewing" find=""
    glyph : text "$d.glyph"
      role appglyph
    label : text "$d.title"
      role applabel
      clamp 1
  chat : column
    only when $d.opens = "chat"
    role appicon
    role appicon.on when ($v.viewing or "") = "" and ($v.find or "") = ""
    on click do prop $v viewing="" back="$v.viewing" find=""
    glyph : text "$d.glyph"
      role appglyph
    label : text "$d.title"
      role applabel
      clamp 1
  window : column
    only when ($d.window or "") != ""
    role appicon
    on click host window $d.window
    glyph : text "$d.glyph"
      role appglyph
    label : text "$d.title"
      role applabel
      clamp 1
adder : column
  role appadd
  on click do create $k.item kind=note title="$r.title" glyph="◆" opens=$v.viewing
  glyph : text "+"
    role appglyph
  label : text "Dock this"
    role applabel
    clamp 1
